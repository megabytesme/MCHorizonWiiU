cmake_minimum_required(VERSION 3.13)
project(horizon_gx2_world_scale_transition C)
option(HORIZON_SCALE_CEMU_READBACK "Enable the Cemu-only CPU pixel readback suite" OFF)
option(HORIZON_SCALE_VISUAL_STORAGE "Compare offscreen color storage and the composite context" OFF)
if(HORIZON_SCALE_CEMU_READBACK AND HORIZON_SCALE_VISUAL_STORAGE)
   message(FATAL_ERROR "Choose one GX2 scale probe mode")
endif()

if(NOT EXISTS "${HORIZON_WIIU_ASSEMBLED}/wiiu/loading_host/world_scale.c")
   message(FATAL_ERROR "Set HORIZON_WIIU_ASSEMBLED to the staged Horizon source")
endif()

add_subdirectory("${HORIZON_WIIU_ASSEMBLED}/wiiu/loading_host" loading_host)
add_executable(gx2_world_scale_transition main.c)
target_compile_options(gx2_world_scale_transition PRIVATE -Wall -Wextra -Werror)
if(HORIZON_SCALE_CEMU_READBACK)
   target_compile_definitions(gx2_world_scale_transition PRIVATE HORIZON_SCALE_CEMU_READBACK)
endif()
if(HORIZON_SCALE_VISUAL_STORAGE)
   target_compile_definitions(gx2_world_scale_transition PRIVATE HORIZON_SCALE_VISUAL_STORAGE)
endif()
target_link_libraries(gx2_world_scale_transition PRIVATE horizon_wiiu_loading_host)
wut_create_rpx(gx2_world_scale_transition)
