cmake_minimum_required(VERSION 3.13)
project(horizon_gx2_world_scale_diagnostic C)
option(HORIZON_SCALE_VISUAL "Present the scale sweep without CPU readback" OFF)
option(HORIZON_SCALE_CLEAR_COMPARISON "Compare world composite with and without output clear" OFF)
option(HORIZON_SCALE_FLIP_COMPARISON "Compare immediate and deferred flip waits" OFF)
if(HORIZON_SCALE_VISUAL AND (HORIZON_SCALE_CLEAR_COMPARISON OR HORIZON_SCALE_FLIP_COMPARISON))
   message(FATAL_ERROR "The presentation comparisons require pixel readback")
endif()

if(NOT EXISTS "${HORIZON_WIIU_ASSEMBLED}/wiiu/loading_host/world_scale.c")
   message(FATAL_ERROR "Set HORIZON_WIIU_ASSEMBLED to the staged Horizon source")
endif()

add_subdirectory("${HORIZON_WIIU_ASSEMBLED}/wiiu/loading_host" loading_host)
add_executable(gx2_world_scale_diagnostic main.c)
target_compile_options(gx2_world_scale_diagnostic PRIVATE -Wall -Wextra -Werror)
if(HORIZON_SCALE_VISUAL)
   target_compile_definitions(gx2_world_scale_diagnostic PRIVATE HORIZON_SCALE_VISUAL)
endif()
if(HORIZON_SCALE_CLEAR_COMPARISON)
   target_compile_definitions(gx2_world_scale_diagnostic PRIVATE HORIZON_SCALE_CLEAR_COMPARISON)
endif()
if(HORIZON_SCALE_FLIP_COMPARISON)
   target_compile_definitions(gx2_world_scale_diagnostic PRIVATE HORIZON_SCALE_FLIP_COMPARISON)
endif()
target_link_libraries(gx2_world_scale_diagnostic PRIVATE horizon_wiiu_loading_host)
wut_create_rpx(gx2_world_scale_diagnostic)
